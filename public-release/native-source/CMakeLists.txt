cmake_minimum_required(VERSION 3.24)
project(QBTC VERSION 0.9.0 LANGUAGES CXX)
option(QBTC_MAINNET "Build the separate experimental mainnet profile" OFF)
option(QBTC_FINAL_PROFILE "Build the isolated frozen final-network candidate" OFF)
if(QBTC_FINAL_PROFILE)
 set(QBTC_MAINNET ON)
 add_compile_definitions(QBTC_FINAL_PROFILE)
endif()
if(QBTC_MAINNET)
 add_compile_definitions(QBTC_MAINNET)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS Crypto)
find_package(Threads REQUIRED)
add_library(qbtc-consensus src/consensus.cpp src/utxo.cpp src/journal.cpp src/storage.cpp src/lock.cpp src/address.cpp)
target_include_directories(qbtc-consensus PUBLIC include)
target_link_libraries(qbtc-consensus PUBLIC OpenSSL::Crypto Threads::Threads)
if(WIN32)
 target_link_libraries(qbtc-consensus PUBLIC advapi32 ws2_32 crypt32)
endif()
if(MSVC)
 target_compile_options(qbtc-consensus PRIVATE /W4 /WX /permissive-)
else()
 target_compile_options(qbtc-consensus PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Wpedantic -Werror)
endif()
add_executable(qbtc src/main.cpp)
target_link_libraries(qbtc PRIVATE qbtc-consensus)
add_executable(qbtc-tests tests/test_consensus.cpp)
target_link_libraries(qbtc-tests PRIVATE qbtc-consensus)
enable_testing()
if(NOT QBTC_MAINNET)
 add_test(NAME consensus COMMAND qbtc-tests)
endif()

add_library(qbtc-wallet-lib src/wallet.cpp)
target_link_libraries(qbtc-wallet-lib PUBLIC qbtc-consensus)
add_executable(qbtc-wallet src/wallet_main.cpp)
target_link_libraries(qbtc-wallet PRIVATE qbtc-wallet-lib)
add_executable(qbtc-peer src/peer_main.cpp src/control.cpp src/net.cpp src/peers.cpp src/mempool.cpp)
target_link_libraries(qbtc-peer PRIVATE qbtc-consensus)
add_executable(qbtc-wallet-tests tests/test_wallet.cpp)
target_link_libraries(qbtc-wallet-tests PRIVATE qbtc-wallet-lib)
if(NOT QBTC_MAINNET)
 add_test(NAME wallet COMMAND qbtc-wallet-tests)
endif()
add_executable(qbtc-crypto-interop tests/crypto_interop.cpp)
target_link_libraries(qbtc-crypto-interop PRIVATE qbtc-wallet-lib)

add_executable(qbtc-journal-tests tests/test_journal.cpp)
target_link_libraries(qbtc-journal-tests PRIVATE qbtc-consensus)
if(NOT QBTC_MAINNET)
 add_test(NAME journal COMMAND qbtc-journal-tests)
endif()
add_executable(qbtc-benchmark tests/benchmark_chain.cpp)
target_compile_definitions(qbtc-benchmark PRIVATE QBTC_DELTAS)
target_link_libraries(qbtc-benchmark PRIVATE qbtc-consensus)

add_executable(qbtc-control-tests tests/test_control.cpp src/control.cpp)
target_link_libraries(qbtc-control-tests PRIVATE qbtc-consensus)
if(NOT QBTC_MAINNET)
 add_test(NAME control COMMAND qbtc-control-tests)
endif()

add_executable(qbtc-fuzz-tests tests/test_fuzz.cpp)
target_link_libraries(qbtc-fuzz-tests PRIVATE qbtc-consensus)
if(NOT QBTC_MAINNET)
 add_test(NAME mutations COMMAND qbtc-fuzz-tests)
endif()

add_executable(qbtc-mempool-tests tests/test_mempool.cpp src/mempool.cpp)
target_link_libraries(qbtc-mempool-tests PRIVATE qbtc-consensus)
if(NOT QBTC_MAINNET)
 add_test(NAME mempool COMMAND qbtc-mempool-tests)
endif()

add_executable(qbtc-network-tests tests/test_internet_policy.cpp src/net.cpp src/peers.cpp)
target_link_libraries(qbtc-network-tests PRIVATE qbtc-consensus)
if(NOT QBTC_MAINNET)
 add_test(NAME network-policy COMMAND qbtc-network-tests)
endif()

add_executable(qbtc-peers-tests tests/test_peers.cpp src/net.cpp src/peers.cpp)
target_link_libraries(qbtc-peers-tests PRIVATE qbtc-consensus)
if(NOT QBTC_MAINNET)
 add_test(NAME peers COMMAND qbtc-peers-tests)
endif()

if(QBTC_MAINNET)
 if(QBTC_FINAL_PROFILE)
  set(QBTC_PROFILE_TEST tests/test_final_network.cpp)
 else()
  set(QBTC_PROFILE_TEST tests/test_mainnet.cpp)
 endif()
 add_executable(qbtc-mainnet-tests ${QBTC_PROFILE_TEST} src/control.cpp src/net.cpp src/peers.cpp src/mempool.cpp)
 target_link_libraries(qbtc-mainnet-tests PRIVATE qbtc-wallet-lib)
 add_test(NAME mainnet-isolation COMMAND qbtc-mainnet-tests)
endif()

add_executable(qbtc-parallel-mining-tests tests/test_parallel_mining.cpp)
target_link_libraries(qbtc-parallel-mining-tests PRIVATE qbtc-consensus)
add_test(NAME parallel-mining COMMAND qbtc-parallel-mining-tests)

add_executable(qbtc-difficulty-launch-tests tests/test_difficulty_launch.cpp)
target_link_libraries(qbtc-difficulty-launch-tests PRIVATE qbtc-consensus)
add_test(NAME difficulty-launch COMMAND qbtc-difficulty-launch-tests)

if(QBTC_MAINNET)
 add_executable(qbtc-reset-v2-tests tests/test_reset_v2.cpp)
 target_link_libraries(qbtc-reset-v2-tests PRIVATE qbtc-consensus)
 add_test(NAME reset-v2-consensus COMMAND qbtc-reset-v2-tests)
endif()

add_executable(qbtc-security-admission-tests tests/test_security_admission.cpp src/mempool.cpp)
target_link_libraries(qbtc-security-admission-tests PRIVATE qbtc-wallet-lib)
add_test(NAME security-admission COMMAND qbtc-security-admission-tests)

add_executable(qbtc-request-budget-tests tests/test_request_budget.cpp)
target_link_libraries(qbtc-request-budget-tests PRIVATE qbtc-consensus)
add_test(NAME request-budget COMMAND qbtc-request-budget-tests)

if(NOT QBTC_MAINNET)
 add_executable(qbtc-node-policy-tests tests/test_node_policy.cpp src/control.cpp src/net.cpp src/peers.cpp src/mempool.cpp)
 target_link_libraries(qbtc-node-policy-tests PRIVATE qbtc-wallet-lib)
 add_test(NAME node-policy COMMAND qbtc-node-policy-tests)
endif()

if(MINGW)
 foreach(target qbtc qbtc-wallet qbtc-peer)
  target_link_options(${target} PRIVATE -municode)
 endforeach()
endif()
